docker compose -p jidan-sandbox -f back-end/testing/compose.yml exec mysql sh -c 'MYSQL_PWD=$MYSQL_PASSWORD mysql -u "$MYSQL_USER" "$MYSQL_DATABASE"'

SELECT * FROM worker_profiles;
